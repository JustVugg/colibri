 * pool meets it (backend_vulkan.c pool_new_block: a block of 256 MiB or 1 GiB with the device-address flag, tried again for every
 * tensor that does not fit): fill the type with 256 MiB blocks until the driver refuses, then ask NTRY more blocks of 256 MiB
 * and NTRY of 1 GiB and print the milliseconds of each refusal; then free everything.
 *   gcc -O1 -Wall -o sonda5 sonda5.c -lvulkan && VK_DRIVER_FILES=<icd> ./sonda5 */
#include <vulkan/vulkan.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <time.h>

#define MAXB 2048
#define NTRY 8

static VkDevice dev;
static VkDeviceMemory mem[MAXB];

static double now(void) { struct timespec t; clock_gettime(CLOCK_MONOTONIC, &t); return t.tv_sec * 1e3 + t.tv_nsec / 1e6; }
static VkResult alloc(uint32_t type, VkDeviceSize bytes, VkDeviceMemory *m) {
    VkMemoryAllocateFlagsInfo fl = {.sType = VK_STRUCTURE_TYPE_MEMORY_ALLOCATE_FLAGS_INFO, .flags = VK_MEMORY_ALLOCATE_DEVICE_ADDRESS_BIT};
    VkMemoryAllocateInfo ai = {.sType = VK_STRUCTURE_TYPE_MEMORY_ALLOCATE_INFO, .pNext = &fl, .allocationSize = bytes, .memoryTypeIndex = type};
    return vkAllocateMemory(dev, &ai, NULL, m);
}
static void tries(uint32_t type, int mib) {
    printf("  %d more blocks of %d MiB, ms each:", NTRY, mib);
    for (int i = 0; i < NTRY; i++) {
        VkDeviceMemory m; double t0 = now();
        VkResult r = alloc(type, (VkDeviceSize)mib << 20, &m);
        printf(" %.0f%s", now() - t0, r == VK_SUCCESS ? "(taken)" : "");
        if (r == VK_SUCCESS) vkFreeMemory(dev, m, NULL);
    }
    printf("\n"); fflush(stdout);
}
