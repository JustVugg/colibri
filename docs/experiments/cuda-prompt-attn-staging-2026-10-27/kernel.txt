## the test program, vm/banco_attn.cu (built from c/ as make cuda-chain-check builds its test: backend_cuda.cu cuda_chain.cu tests/banco_attn.cu)
/* banco_attn.cu: the prompt attention of Qwen3.6's CUDA chain on the engine's shapes, for the plan of
 * model-inspection/pr-cuda-prompt-attn-20261027 (step 1, the diagnosis; step 3, the prototype). It takes the engine's
 * own entry (T->attn = cc_attn, which runs cc_attn_fk<256> from 16 rows) with the fields qwen36_cuda_chain.h:932-935
 * fills: 2048 rows after 0, 2048, 4096 and 6144 positions (the four chunks of an 8192 prompt), 16 query heads on 2 kv
 * heads of 256, the output gate beside each head's query (the test's "prompt rows" case), the cache's capacity 8384
 * (the bench's Q36_MAXT at 8192). Built like `make cuda-chain-check` (c/Makefile:1333), from c/:
 * backend_cuda.cu cuda_chain.cu tests/banco_attn.cu.
 *   banco_attn [reps [shape]]   reps timed calls a shape after one warm-up (default 5); shape 0..3, or -1 for all
 * A call is begin, attn, submit(1) (submit waits for the device), timed on the host's clock. Each shape then checks
 * rows 0, 1, 1024 and 2047 against a reference in double over all their positions, and prints a hash of the whole
 * output: two kernels that agree bit for bit print the same hash. */
#include "../backend_cuda.h"
#include "../cuda_chain.h"
#include <algorithm>
#include <chrono>
#include <cmath>
#include <cstdint>
#include <cstdio>
#include <cstdlib>
#include <vector>

static const ColiCudaChainOps *T;
static float frand(unsigned *s) { *s = *s * 1664525u + 1013904223u; return ((*s >> 8) & 0xFFFF) / 65535.f * 2.f - 1.f; }
static std::vector<float> rnd(size_t n, unsigned seed) { std::vector<float> v(n); unsigned s = seed; for (auto &x : v) x = frand(&s); return v; }
static CcBuf *up(const std::vector<float> &v) {
    CcBuf *b = T->buf(v.size() * 4, CC_DEV);
    if (!b || !T->begin() || !T->write(b, 0, v.data(), v.size() * 4) || !T->submit(1)) { printf("upload failed\n"); exit(1); }
    return b;
}
/* one row's output, every head, in double (the test's ref_attn, no selection list) */
static void ref_row(const float *q, const float *kc, const float *vc, const CcAttn *p, int s, std::vector<double> &out) {
    int rep = p->H / p->KVH, n = p->pos_base + s + 1;
    std::vector<double> e(n);
    out.assign((size_t)p->H * p->hd, 0.0);
    for (int h = 0; h < p->H; h++) {
        const float *qs = q + p->q_off + (size_t)s * p->q_row + h * p->q_seg, *g = q + p->g_off + (size_t)s * p->g_row + h * p->g_seg;
        size_t kb = (size_t)(h / rep) * p->cap;
        double m = -1e300, l = 0, *o = &out[(size_t)h * p->hd];
        for (int t = 0; t < n; t++) {
            const float *k = kc + p->k_off + (kb + t) * p->hd;
            double a = 0; for (int d = 0; d < p->hd; d++) a += (double)qs[d] * k[d];
            e[t] = a * p->scale; m = std::max(m, e[t]);
        }
        for (int t = 0; t < n; t++) { e[t] = exp(e[t] - m); l += e[t]; }
        for (int t = 0; t < n; t++) { const float *v = vc + p->v_off + (kb + t) * p->hd; for (int d = 0; d < p->hd; d++) o[d] += e[t] * v[d]; }
        for (int d = 0; d < p->hd; d++) o[d] = o[d] / l / (1.0 + exp(-(double)g[d]));
    }
}
static uint64_t fnv(const void *b, size_t n) {   /* FNV-1a over the output's bytes */
    const unsigned char *c = (const unsigned char *)b; uint64_t h = 1469598103934665603ull;
    for (size_t i = 0; i < n; i++) { h ^= c[i]; h *= 1099511628211ull; }
    return h;
}

int main(int argc, char **argv) {
    int reps = argc > 1 ? atoi(argv[1]) : 5, only = argc > 2 ? atoi(argv[2]) : -1;
    int dev = 0;
    if (!coli_cuda_init(&dev, 1)) { printf("no CUDA device\n"); return 1; }
    T = coli_cuda_chain_ops();
    if (!T || !T->init(0)) { printf("no chain context\n"); return 1; }
    const int H = 16, KV = 2, HD = 256, S = 2048, CAP = 8384, qdim = 2 * HD;   /* a head's query, then its gate */
    std::vector<float> q = rnd((size_t)S * H * qdim, 51), kc = rnd((size_t)KV * CAP * HD, 52), vc = rnd((size_t)KV * CAP * HD, 53);
    std::vector<float> o((size_t)S * H * HD);
    CcBuf *qb = up(q), *kb = up(kc), *vb = up(vc), *ob = T->buf(o.size() * 4, CC_DEV);
    const int rows[4] = {0, 1, S / 2, S - 1};
    printf("chunk  after  calls  median ms  best ms  TFLOP/s  max|diff| rows 0,1,%d,%d  output hash\n", S / 2, S - 1);
    for (int c = 0; c < 4; c++) {
        if (only >= 0 && c != only) continue;
        int pb = 2048 * c;
        CcAttn p = {S, H, KV, HD, pb, CAP, 0, H * qdim, qdim, HD, H * qdim, qdim, 1, 0, H * HD, 0, 0, 1.f / sqrtf((float)HD), 0, 0};
        std::vector<double> ms;
        for (int r = 0; r <= reps; r++) {   /* r 0 is the warm-up */
            auto t0 = std::chrono::steady_clock::now();
            if (!(T->begin() && T->attn(qb, kb, vb, ob, qb, NULL, &p) && T->submit(1))) { printf("attn failed\n"); return 1; }
            double t = std::chrono::duration<double, std::milli>(std::chrono::steady_clock::now() - t0).count();
            if (r) ms.push_back(t);
        }
        if (!T->read(ob, 0, o.data(), o.size() * 4)) { printf("read back failed\n"); return 1; }
        double d = 0;
        std::vector<double> ref;
        for (int s : rows) {
            ref_row(q.data(), kc.data(), vc.data(), &p, s, ref);
            for (size_t i = 0; i < ref.size(); i++) d = std::max(d, fabs(ref[i] - o[(size_t)s * H * HD + i]));
        }
        std::vector<double> sm = ms;
        std::sort(sm.begin(), sm.end());
        double med = sm[sm.size() / 2], pairs = (double)S * pb + (double)S * (S + 1) / 2;
        printf("%5d  %5d  %5d  %9.3f  %7.3f  %7.2f  %.2e  %016llx\n", c, pb, reps, med, sm[0],
               pairs * 2 * 2 * H * HD / (med / 1e3) / 1e12, d, (unsigned long long)fnv(o.data(), o.size() * 4));
        printf("       calls:");
        for (double t : ms) printf(" %.3f", t);
        printf("\n");
    }
    T->free(qb); T->free(kb); T->free(vb); T->free(ob);
    return 0;
}
