## The scripts that read the raw files

### riassunto_orologi.py
#!/usr/bin/env python3
"""riassunto_orologi.py <dir>: for each N-<version>.smi of fase-a.sh's step 11 (nvidia-smi samples beside the 512-token prompt alone),
the samples above 45 W in the last 5 s before the process ended (N-<version>.t1): how many, over how long, the SM clock (mean, min,
max, MHz), the power (mean, max, W), the temperature (C) and how many samples are at 69 W or more; then the spacing of the samples."""
import csv, datetime, glob, os, re, statistics, sys
num = lambda s: float(re.findall(r"[-+]?\d+\.?\d*", s)[0])
for smi in sorted(glob.glob(os.path.join(sys.argv[1], "*.smi"))):
    base = smi[:-4]
    t1 = float(open(base + ".t1").read())
    c = list(csv.DictReader(open(base + ".csv")))[-1]
    rows = []
    for x in list(csv.reader(open(smi)))[1:]:
        ts = datetime.datetime.strptime(x[0].strip(), "%Y/%m/%d %H:%M:%S.%f").replace(tzinfo=datetime.timezone.utc).timestamp()
        rows.append((ts - t1, num(x[1]), num(x[3]), num(x[4])))
    w = [r for r in rows if -5 <= r[0] <= 0 and r[2] >= 45]
    gaps = [(b[0] - a[0]) * 1000 for a, b in zip(rows, rows[1:])]
    print(f"{os.path.basename(base)}: prefill_s {c['prefill_s']}; {len(w)} samples above 45 W in the last 5 s over {w[-1][0] - w[0][0]:.2f} s; "
          f"SM clock mean {sum(r[1] for r in w) / len(w):.0f} min {min(r[1] for r in w):.0f} max {max(r[1] for r in w):.0f} MHz; "
          f"power mean {sum(r[2] for r in w) / len(w):.1f} max {max(r[2] for r in w):.1f} W; temp {max(r[3] for r in w):.0f} C; "
          f"{sum(1 for r in w if r[2] >= 69)} samples at 69 W or more; {len(rows)} samples, median spacing {statistics.median(gaps):.0f} ms")

### analizza_orologi.py
#!/usr/bin/env python3
"""analizza_orologi.py <dir> [secondi]: the clocks during the 512 alone (fase-a.sh step 11). For each N-<versione>.smi
(nvidia-smi every 50 ms: timestamp, clocks.sm, clocks.mem, power.draw, temperature.gpu[, throttle reasons]) it prints,
for the last <secondi> (default 6) before the process ended (N-<versione>.t1), the bins of 0.5 s: mean and minimum
SM clock (MHz), mean power (W), temperature, throttle reasons seen. The csv of the run says how long the prompt took."""
import csv, re, sys, os, glob, datetime

d = sys.argv[1]
tail = float(sys.argv[2]) if len(sys.argv) > 2 else 6.0
num = lambda s: float(re.findall(r"[-+]?\d+\.?\d*", s)[0])

def parse(path):
    rows = []
    with open(path) as f:
        r = csv.reader(f)
        head = next(r)
        for x in r:
            if len(x) < 5:
                continue
            ts = datetime.datetime.strptime(x[0].strip(), "%Y/%m/%d %H:%M:%S.%f").replace(tzinfo=datetime.timezone.utc).timestamp()
            rows.append((ts, num(x[1]), num(x[2]), num(x[3]), num(x[4]), x[5].strip() if len(x) > 5 else ""))
    return head, rows

for smi in sorted(glob.glob(os.path.join(d, "*.smi"))):
    base = smi[:-4]
    t0 = float(open(base + ".t0").read()); t1 = float(open(base + ".t1").read())
    c = list(csv.DictReader(open(base + ".csv")))[-1]
    head, rows = parse(smi)
    print(f"== {os.path.basename(base)}: {t1 - t0:.1f} s, prefill {c['prefill_s']} s ({c['prefill_tps']} tok/s), first token {c['gen_first_ms']} ms, "
          f"{len(rows)} samples, first {rows[0][0] - t0:+.1f} s, last {rows[-1][0] - t1:+.1f} s from the run's ends")
    lo = t1 - tail
    bins = {}
    for ts, sm, mem, w, tmp, th in rows:
        if ts < lo or ts > t1 + 1:
            continue
        bins.setdefault(int((ts - lo) / 0.5), []).append((sm, w, tmp, th))
    print("   t-t1(s)   sm mean/min   power W   temp  throttle")
    for b in sorted(bins):
        v = bins[b]
        print(f"   {lo + b * 0.5 - t1:+6.1f}   {sum(x[0] for x in v) / len(v):7.0f}/{min(x[0] for x in v):4.0f}   {sum(x[1] for x in v) / len(v):7.1f}   {max(x[2] for x in v):4.0f}  {','.join(sorted(set(x[3] for x in v)))}")

### analizza_foto.py
#!/usr/bin/env python3
"""analizza_foto.py <p.sqlite> <p.steps>: the prompt of an nsys run of the bench (one chunk, 40 layers), window anchored on the
MoE kernels: from the first hcg_moe_gu_k to the end of the last hcg_moe_dn_k (the warm start's copies come before and the
idle-gap rule of analizza_nsys.py cannot cut them off here). In it: how long the card was busy with kernels, with copies, with
both at once, the HtoD bytes, and the layers' period (the start of one MoE to the next), so the median layer."""
import sqlite3, sys
from statistics import median

db, steps = sqlite3.connect(sys.argv[1]), sys.argv[2]
names = dict(db.execute("SELECT id, value FROM StringIds"))
K = sorted((s, e, names.get(n, str(n))) for s, e, n in db.execute("SELECT start, end, shortName FROM CUPTI_ACTIVITY_KIND_KERNEL"))
C = sorted(db.execute("SELECT start, end, bytes, copyKind FROM CUPTI_ACTIVITY_KIND_MEMCPY"))
gu = [k for k in K if k[2] == "hcg_moe_gu_k"]; dn = [k for k in K if k[2] == "hcg_moe_dn_k"]
t0 = gu[-40][0]; t1 = dn[-1][1]
ms = lambda ns: ns / 1e6
prompt = [float(l.split()[4]) for l in open(steps) if l.split()[1] == "-1"][0]
def union(iv):
    iv = sorted(iv); out = []
    for s, e in iv:
        if out and s <= out[-1][1]: out[-1][1] = max(out[-1][1], e)
        else: out.append([s, e])
    return out
def length(u): return sum(e - s for s, e in u)
def inter(a, b):
    i = j = tot = 0
    while i < len(a) and j < len(b):
        s, e = max(a[i][0], b[j][0]), min(a[i][1], b[j][1])
        if e > s: tot += e - s
        if a[i][1] < b[j][1]: i += 1
        else: j += 1
    return tot
def clip(iv): return [(max(s, t0), min(e, t1)) for s, e in iv if e > t0 and s < t1]
ku = union(clip([(s, e) for s, e, _ in K])); cu = union(clip([(s, e) for s, e, b, kind in C if kind == 1]))
hb = sum(b for s, e, b, kind in C if kind == 1 and s >= t0 and s < t1)
per = [ms(gu[-40 + i + 1][0] - gu[-40 + i][0]) for i in range(39)]
print(f"steps file: prompt {prompt:.1f} ms; MoE span {ms(t1 - t0):.1f} ms (first hcg_moe_gu_k to the last hcg_moe_dn_k end)")
print(f"in the span: kernels {ms(length(ku)):.1f} ms, HtoD copies {ms(length(cu)):.1f} ms, both at once {ms(inter(ku, cu)):.1f} ms; "
      f"{hb / 1e9:.2f} GB copied ({hb / 1e9 / ms(length(cu)) * 1e3:.1f} GB/s while copying)")
print(f"layer period (first MoE start to the next): median {median(per):.1f} ms, mean {sum(per) / len(per):.1f}, min {min(per):.1f}, max {max(per):.1f}")
